     1	const CACHE_NAME = 'textshow-v5';
     2	const ASSETS = [
     3	  './',
     4	  './index.html',
     5	  './css/style.css',
     6	  './js/app.js',
     7	  './manifest.webmanifest',
     8	  './icons/favicon.svg',
     9	  './icons/icon-180.png',
    10	  './icons/icon-192.png',
    11	  './icons/icon-512.png'
    12	];
    13	
    14	self.addEventListener('install', (event) => {
    15	  event.waitUntil(
    16	    caches
    17	      .open(CACHE_NAME)
    18	      .then((cache) => cache.addAll(ASSETS))
    19	      .then(() => self.skipWaiting())
    20	  );
    21	});
    22	
    23	self.addEventListener('activate', (event) => {
    24	  event.waitUntil(
    25	    caches
    26	      .keys()
    27	      .then((keys) => Promise.all(keys.filter((key) => key !== CACHE_NAME).map((key) => caches.delete(key))))
    28	      .then(() => self.clients.claim())
    29	  );
    30	});
    31	
    32	self.addEventListener('fetch', (event) => {
    33	  if (event.request.method !== 'GET') return;
    34	
    35	  event.respondWith(
    36	    caches.match(event.request).then((cached) => {
    37	      if (cached) return cached;
    38	      return fetch(event.request)
    39	        .then((response) => {
    40	          const copy = response.clone();
    41	          caches.open(CACHE_NAME).then((cache) => cache.put(event.request, copy));
    42	          return response;
    43	        })
    44	        .catch(() => caches.match('./index.html'));
    45	    })
    46	  );
    47	});

   432	  const torch = {
   433	    stream: null,
   434	    track: null,
   435	    timer: null,
   436	    lit: false,
   437	    active: false,
   438	
   439	    async start(hz) {
   440	      if (!window.isSecureContext) {
   441	        throw new Error('需要 HTTPS 或 localhost 才能调用闪光灯');
   442	      }
   443	      if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
   444	        throw new Error('当前浏览器不支持摄像头调用');
   445	      }
   446	
   447	      if (!this.stream) {
   448	        let stream = null;
   449	        try {
   450	          stream = await navigator.mediaDevices.getUserMedia({
   451	            video: { facingMode: { ideal: 'environment' } },
   452	            audio: false
   453	          });
   454	        } catch (error) {
   455	          if (error && error.name === 'NotAllowedError') {
   456	            throw new Error('未获得摄像头权限，请在浏览器设置中允许后重试');
   457	          }
   458	          if (error && error.name === 'NotFoundError') {
   459	            throw new Error('未检测到可用的摄像头');
   460	          }
   461	          throw new Error('无法打开摄像头，请检查权限');
   462	        }
   463	        this.stream = stream;
   464	        this.track = stream.getVideoTracks()[0];
   465	        const video = $('#torchVideo');
   466	        video.srcObject = stream;
   467	        try {
   468	          await video.play();
   469	        } catch (error) {
   470	          /* hidden video may not need playback */
   471	        }
   472	      }
   473	
   474	      const capabilities = this.track.getCapabilities ? this.track.getCapabilities() : {};
   475	      if (!capabilities || !capabilities.torch) {
   476	        await this.stop();
   477	        throw new Error('此浏览器不支持手电筒控制，iOS 请使用系统手电筒');
   478	      }
   479	
   480	      this.active = true;
   481	      this.lit = true;
   482	      await this.applyTorch(true);
   483	      this.restart(hz);
   484	    },
   485	
   486	    async applyTorch(on) {
   487	      if (!this.track) return;
   488	      try {
   489	        await this.track.applyConstraints({ advanced: [{ torch: on }] });
   490	      } catch (error) {
   491	        /* some devices reject rapid constraint changes */
   492	      }
   493	    },
   494	
   495	    restart(hz) {
   496	      if (this.timer) {
   497	        clearInterval(this.timer);
   498	        this.timer = null;
   499	      }
   500	      const halfPeriod = Math.max(20, 500 / hz);
   501	      this.timer = setInterval(() => {
   502	        if (!this.active) return;
   503	        this.lit = !this.lit;
   504	        this.applyTorch(this.lit);
   505	      }, halfPeriod);
   506	    },
   507	
   508	    async stop() {
   509	      this.active = false;
   510	      if (this.timer) {
   511	        clearInterval(this.timer);
   512	        this.timer = null;
   513	      }
   514	      if (this.track) {
   515	        await this.applyTorch(false);
   516	      }
   517	      this.track = null;
   518	      if (this.stream) {
   519	        this.stream.getTracks().forEach((track) => {
   520	          try {
   521	            track.stop();
   522	          } catch (error) {
   523	            /* ignore */
   524	          }
   525	        });
   526	      }
   527	      this.stream = null;
   528	      const video = $('#torchVideo');
   529	      try {
   530	        video.srcObject = null;
   531	      } catch (error) {
   532	        /* ignore */
   533	      }
   534	    }
   535	  };

   538	
   539	  class SoundEngine {
   540	    constructor() {
   541	      this.ctx = null;
   542	      this.master = null;
   543	      this.timer = null;
   544	      this.nodes = [];
   545	      this.active = false;
   546	    }
   547	
   548	    async init() {
   549	      if (!this.ctx) {
   550	        const AudioContextClass = window.AudioContext || window.webkitAudioContext;
   551	        if (!AudioContextClass) throw new Error('当前浏览器不支持音频播放');
   552	        this.ctx = new AudioContextClass();
   553	        const compressor = this.ctx.createDynamicsCompressor();
   554	        compressor.threshold.value = -16;
   555	        compressor.knee.value = 20;
   556	        compressor.ratio.value = 8;
   557	        compressor.attack.value = 0.003;
   558	        compressor.release.value = 0.15;
   559	        this.master = this.ctx.createGain();
   560	        this.master.gain.value = 0;
   561	        this.master.connect(compressor);
   562	        compressor.connect(this.ctx.destination);
   563	      }
   564	      if (this.ctx.state === 'suspended') {
   565	        await this.ctx.resume();
   566	      }
   567	    }
   568	
   569	    level(volume) {
   570	      return clamp(Math.pow(clamp(volume, 0, 1), 1.7) * 0.9, 0, 1);
   571	    }
   572	
   573	    async start(pattern, volume) {
   574	      await this.init();
   575	      this.halt();
   576	      this.active = true;
   577	      const now = this.ctx.currentTime;
   578	      this.master.gain.cancelScheduledValues(now);
   579	      this.master.gain.setValueAtTime(this.level(volume), now);
   580	
   581	      if (pattern === 'siren') {
   582	        this.startSiren();
   583	      } else if (pattern === 'steady') {
   584	        this.startSteady();
   585	      } else {
   586	        this.startBeeps(pattern === 'urgent');
   587	      }
   588	    }
   589	
   590	    halt() {
   591	      this.active = false;
   592	      if (this.timer) {
   593	        clearInterval(this.timer);
   594	        this.timer = null;
   595	      }
   596	      this.nodes.forEach((node) => {
   597	        if (typeof node.stop === 'function') {
   598	          try {
   599	            node.stop();
   600	          } catch (error) {
   601	            /* ignore */
   602	          }
   603	        }
   604	        try {
   605	          node.disconnect();
   606	        } catch (error) {
   607	          /* ignore */
   608	        }
   609	      });
   610	      this.nodes = [];
   611	      if (this.ctx && this.master) {
   612	        const now = this.ctx.currentTime;
   613	        this.master.gain.cancelScheduledValues(now);
   614	        this.master.gain.setValueAtTime(0, now);
   615	      }
   616	    }
   617	
   618	    stop() {
   619	      this.halt();
   620	    }
   621	
   622	    setVolume(volume) {
   623	      if (!this.ctx || !this.master) return;
   624	      const now = this.ctx.currentTime;
   625	      this.master.gain.cancelScheduledValues(now);
   626	      this.master.gain.setTargetAtTime(this.level(volume), now, 0.04);
   627	    }
   628	
   629	    beep(frequency, duration) {
   630	      const ctx = this.ctx;
   631	      const now = ctx.currentTime;
   632	      const osc = ctx.createOscillator();
   633	      const filter = ctx.createBiquadFilter();
   634	      const gain = ctx.createGain();
   635	
   636	      osc.type = 'square';
   637	      osc.frequency.setValueAtTime(frequency, now);
   638	
   639	      filter.type = 'lowpass';
   640	      filter.frequency.setValueAtTime(Math.min(7000, frequency * 2.2), now);
   641	
   642	      gain.gain.setValueAtTime(0.0001, now);
   643	      gain.gain.linearRampToValueAtTime(0.16, now + 0.006);
   644	      gain.gain.setValueAtTime(0.16, now + Math.max(0.01, duration - 0.015));
   645	      gain.gain.exponentialRampToValueAtTime(0.0001, now + duration);
   646	
   647	      osc.connect(filter);
   648	      filter.connect(gain);
   649	      gain.connect(this.master);
   650	      osc.start(now);
   651	      osc.stop(now + duration + 0.02);
   652	      osc.onended = () => {
   653	        try {
   654	          osc.disconnect();
   655	          filter.disconnect();
   656	          gain.disconnect();
   657	        } catch (error) {
   658	          /* ignore */
   659	        }
   660	      };
   661	    }
   662	
   663	    startBeeps(urgent) {
   664	      const frequency = urgent ? 3100 : 2600;
   665	      const duration = urgent ? 0.07 : 0.16;
   666	      const period = urgent ? 0.15 : 0.42;
   667	      const tick = () => {
   668	        if (this.active) this.beep(frequency, duration);
   669	      };
   670	      tick();
   671	      this.timer = setInterval(tick, period * 1000);
   672	    }
   673	
   674	    startSiren() {
   675	      const ctx = this.ctx;
   676	      const osc = ctx.createOscillator();
   677	      const filter = ctx.createBiquadFilter();
   678	      const gain = ctx.createGain();
   679	
   680	      osc.type = 'triangle';
   681	      filter.type = 'lowpass';
   682	      filter.frequency.value = 5000;
   683	      gain.gain.value = 0.13;
   684	
   685	      let rising = true;
   686	      const sweep = () => {
   687	        const now = ctx.currentTime;
   688	        osc.frequency.cancelScheduledValues(now);
   689	        osc.frequency.setValueAtTime(rising ? 750 : 1750, now);
   690	        osc.frequency.exponentialRampToValueAtTime(rising ? 1750 : 750, now + 0.62);
   691	        rising = !rising;
   692	      };
   693	
   694	      osc.connect(filter);
   695	      filter.connect(gain);
   696	      gain.connect(this.master);
   697	      sweep();
   698	      osc.start();
   699	      this.nodes.push(osc, gain, filter);
   700	      this.timer = setInterval(sweep, 660);
   701	    }
   702	
   703	    startSteady() {
   704	      const ctx = this.ctx;
   705	      const osc = ctx.createOscillator();
   706	      const filter = ctx.createBiquadFilter();
   707	      const gain = ctx.createGain();
   708	      const lfo = ctx.createOscillator();
   709	      const lfoGain = ctx.createGain();
   710	
   711	      osc.type = 'square';
   712	      osc.frequency.value = 2350;
   713	      filter.type = 'lowpass';
   714	      filter.frequency.value = 6000;
   715	      gain.gain.value = 0.09;
   716	
   717	      lfo.type = 'sine';
   718	      lfo.frequency.value = 6;
   719	      lfoGain.gain.value = 0.035;
   720	
   721	      osc.connect(filter);
   722	      filter.connect(gain);
   723	      gain.connect(this.master);
   724	      lfo.connect(lfoGain);
   725	      lfoGain.connect(gain.gain);
   726	
   727	      osc.start();
   728	      lfo.start();
   729	      this.nodes.push(osc, gain, filter, lfo, lfoGain);
   730	    }
   731	  }

   735	  /* ---------- Three-stripe alert ---------- */
   736	  // Cycle: stripe0 on → gap → stripe1 on → gap → stripe2 on → gap → loop
   737	  // Each lit phase ~240ms, gap ~70ms → full cycle ~930ms (~1.08 Hz)
   738	  const alertStripes = {
   739	    timer: null,
   740	    active: false,
   741	    // 0=red, 1=amber, 2=white, 3=gap (all off faint edge only)
   742	    seq: [0, 3, 1, 3, 2, 3],
   743	    step: 0,
   744	
   745	    start() {
   746	      if (this.active) return;
   747	      this.active = true;
   748	      stage.classList.add('alert');
   749	      this.step = 0;
   750	      this.tick();
   751	      this.timer = setInterval(() => this.tick(), 240);
   752	    },
   753	
   754	    tick() {
   755	      const idx = this.seq[this.step % this.seq.length];
   756	      stage.setAttribute('data-strip', String(idx));
   757	      this.step += 1;
   758	    },
   759	
   760	    stop() {
   761	      if (this.timer) {
   762	        clearInterval(this.timer);
   763	        this.timer = null;
   764	      }
   765	      this.active = false;
   766	      stage.classList.remove('alert');
   767	      stage.removeAttribute('data-strip');
   768	    }
