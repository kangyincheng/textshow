1957        this.master = null;
1958        this.timer = null;
1959        this.nodes = [];

1963      async init() {
1964        if (!this.ctx) {
1965          const AudioContextClass = window.AudioContext || window.webkitAudioContext;
1966          if (!AudioContextClass) throw new Error('当前浏览器不支持音频播放');
1967          this.ctx = new AudioContextClass();
1968          const compressor = this.ctx.createDynamicsCompressor();
1969          compressor.threshold.value = -16;
1970          compressor.knee.value = 20;
1971          compressor.ratio.value = 8;
1972          compressor.attack.value = 0.003;
1973          compressor.release.value = 0.15;
1974          this.master = this.ctx.createGain();
1975          this.master.gain.value = 0;
1976          this.master.connect(compressor);
1977          compressor.connect(this.ctx.destination);
1978        }
1979        if (this.ctx.state === 'suspended') {
1980          await this.ctx.resume();
1981        }
1982      }
1983  
1984      level(volume) {
1985        return clamp(Math.pow(clamp(volume, 0, 1), 1.7) * 0.9, 0, 1);
1986      }
1987  
1988      async start(pattern, volume) {
1989        await this.init();
1990        this.halt();
1991        this.active = true;
1992        const now = this.ctx.currentTime;
1993        this.master.gain.cancelScheduledValues(now);
1994        this.master.gain.setValueAtTime(this.level(volume), now);
1995  
1996        if (pattern === 'siren') {
1997          this.startSiren();
1998        } else if (pattern === 'steady') {
1999          this.startSteady();
2000        } else {
2001          this.startBeeps(pattern === 'urgent');
2002        }
2003      }

2005      halt() {
2006        this.active = false;
2007        if (this.timer) {
2008          clearInterval(this.timer);
2009          this.timer = null;
2010        }
2011        this.nodes.forEach((node) => {
2012          if (typeof node.stop === 'function') {
2013            try {
2014              node.stop();
2015            } catch (error) {
2016              /* ignore */
2017            }
2018          }
2019          try {
2020            node.disconnect();
2021          } catch (error) {
2022            /* ignore */
2023          }
2024        });
2025        this.nodes = [];
2026        if (this.ctx && this.master) {
2027          const now = this.ctx.currentTime;
2028          this.master.gain.cancelScheduledValues(now);
2029          this.master.gain.setValueAtTime(0, now);
2030        }
2031      }

2035      }

2038        if (!this.ctx || !this.master) return;
2039        const now = this.ctx.currentTime;
2040        this.master.gain.cancelScheduledValues(now);
2041        this.master.gain.setTargetAtTime(this.level(volume), now, 0.04);

2044      beep(frequency, duration) {
2045        const ctx = this.ctx;
2046        const now = ctx.currentTime;
2047        const osc = ctx.createOscillator();
2048        const filter = ctx.createBiquadFilter();
2049        const gain = ctx.createGain();
2050  
2051        osc.type = 'square';
2052        osc.frequency.setValueAtTime(frequency, now);

2054        filter.type = 'lowpass';
2055        filter.frequency.setValueAtTime(Math.min(7000, frequency * 2.2), now);
2056  
2057        gain.gain.setValueAtTime(0.0001, now);
2058        gain.gain.linearRampToValueAtTime(0.16, now + 0.006);
2059        gain.gain.setValueAtTime(0.16, now + Math.max(0.01, duration - 0.015));
2060        gain.gain.exponentialRampToValueAtTime(0.0001, now + duration);
2061  
2062        osc.connect(filter);
2063        filter.connect(gain);
2064        gain.connect(this.master);
2065        osc.start(now);
2066        osc.stop(now + duration + 0.02);
2067        osc.onended = () => {
2068          try {
2069            osc.disconnect();
2070            filter.disconnect();
2071            gain.disconnect();
2072          } catch (error) {
2073            /* ignore */
2074          }
2075        };
2076      }

2078      startBeeps(urgent) {
2079        const frequency = urgent ? 3100 : 2600;
2080        const duration = urgent ? 0.07 : 0.16;
2081        const period = urgent ? 0.15 : 0.42;
2082        const tick = () => {
2083          if (this.active) this.beep(frequency, duration);
2084        };
2085        tick();

2091        const osc = ctx.createOscillator();
2092        const filter = ctx.createBiquadFilter();
2093        const gain = ctx.createGain();
2094  
2095        osc.type = 'triangle';
2096        filter.type = 'lowpass';
2097        filter.frequency.value = 5000;
2098        gain.gain.value = 0.13;
2099  
2100        let rising = true;
2101        const sweep = () => {
2102          const now = ctx.currentTime;
2103          osc.frequency.cancelScheduledValues(now);
2104          osc.frequency.setValueAtTime(rising ? 750 : 1750, now);
2105          osc.frequency.exponentialRampToValueAtTime(rising ? 1750 : 750, now + 0.62);
2106          rising = !rising;
2107        };
2108  
2109        osc.connect(filter);
2110        filter.connect(gain);
2111        gain.connect(this.master);
2112        sweep();
2113        osc.start();
